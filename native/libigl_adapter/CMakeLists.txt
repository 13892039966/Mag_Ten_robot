cmake_minimum_required(VERSION 3.22)
project(LibiglSofaAdapter LANGUAGES CXX)
find_package(Sofa.Config REQUIRED)
find_package(Eigen3 CONFIG REQUIRED)
add_subdirectory(source)
target_sources(SofaDistanceGrid PRIVATE HybridExactContact.cpp)
target_include_directories(SofaDistanceGrid PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../vendor/libigl/include)
target_link_libraries(SofaDistanceGrid PRIVATE Eigen3::Eigen)
target_compile_features(SofaDistanceGrid PRIVATE cxx_std_17)

target_compile_definitions(SofaDistanceGrid PRIVATE NOMINMAX IGL_PARALLEL_FOR_FORCE_SERIAL)
if(MSVC)
 target_compile_options(SofaDistanceGrid PRIVATE /utf-8)
endif()
if(WIN32)
 target_link_libraries(SofaDistanceGrid PRIVATE bcrypt)
else()
 find_package(OpenSSL REQUIRED)
 target_link_libraries(SofaDistanceGrid PRIVATE OpenSSL::Crypto)
endif()
